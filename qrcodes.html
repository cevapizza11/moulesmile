function genererToken(){
const bytes=new Uint8Array(9);
crypto.getRandomValues(bytes);
return Array.from(bytes,b=>b.toString(36).padStart(2,'0')).join('').slice(0,12);
}
async function prochainNumeroDepart(){
// Regarde le plus grand "numero" déjà en base pour ne jamais recréer de doublon
// même si on relance une génération plus tard pour agrandir le stock.
const snap=await db.collection('jetons').orderBy('numero','desc').limit(1).get();
if(snap.empty) return 1;
return (snap.docs[0].data().numero||0)+1;
}
async function genererEtEnregistrer(){
const btn=document.getElementById('btnGenerer');
const baseUrl=document.getElementById('baseUrl').value.trim();
const nb=parseInt(document.getElementById('nbJetons').value)||150;
if(!baseUrl){alert('Merci de renseigner l\'URL de votre app');return;}
localStorage.setItem('ms_qr_baseUrl',baseUrl);
btn.disabled=true;
const statutBar=document.getElementById('statutBar');
statutBar.textContent='⏳ Génération en cours...';
const depart=await prochainNumeroDepart();
const jetons=[];
for(let i=0;i<nb;i++){
jetons.push({numero:depart+i, token:genererToken()});
}
// Écriture par lots de 400 (limite Firestore = 500 par batch)
try{
for(let i=0;i<jetons.length;i+=400){
const batch=db.batch();
jetons.slice(i,i+400).forEach(j=>{
const ref=db.collection('jetons').doc(j.token);
batch.set(ref,{numero:j.numero, token:j.token, statut:'libre', table:null});
});
await batch.commit();
}
}catch(e){
console.error(e);
statutBar.textContent='❌ Erreur d\'écriture Firestore — voir console';
btn.disabled=false;
return;
}
statutBar.textContent=`✅ ${jetons.length} jetons créés dans Firestore (n°${depart} à n°${depart+jetons.length-1})`;
afficherGrille(jetons, baseUrl);
btn.disabled=false;
}
function afficherGrille(jetons, baseUrl){
const grid=document.getElementById('qrGrid');
grid.innerHTML='';
jetons.forEach(j=>{
const url=`${baseUrl}?j=${j.token}`;
const card=document.createElement('div');
card.className='qr-card';
card.innerHTML=`
<div class="jeton-num">🎫 Jeton n°${j.numero}</div>
<div id="qr_${j.numero}"></div>
<div class="sub">Ne pas afficher le lien complet</div>
`;
grid.appendChild(card);
new QRCode(document.getElementById(`qr_${j.numero}`), {
text: url,
width: 120,
height: 120,
colorDark: "#1A1A1A",
colorLight: "#ffffff",
correctLevel: QRCode.CorrectLevel.H
});
});
